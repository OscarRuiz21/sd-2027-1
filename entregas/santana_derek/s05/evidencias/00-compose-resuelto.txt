name: s05_santana_derek
services:
  app:
    build:
      context: /tmp/mexi-banco-s05-derek
      dockerfile: Dockerfile
    depends_on:
      db:
        condition: service_healthy
        required: true
    environment:
      DB_HOST: db
      DB_NAME: mexibanco
      DB_PASSWORD: mexibanco
      DB_PORT: "5432"
      DB_USER: mexibanco
    healthcheck:
      test:
        - CMD-SHELL
        - wget -qO- http://localhost:8080/actuator/health | grep -q UP || exit 1
      timeout: 3s
      interval: 10s
      retries: 5
    networks:
      default: null
    ports:
      - mode: ingress
        host_ip: 127.0.0.1
        target: 8080
        published: "8085"
        protocol: tcp
  db:
    environment:
      POSTGRES_DB: mexibanco
      POSTGRES_PASSWORD: mexibanco
      POSTGRES_USER: mexibanco
    healthcheck:
      test:
        - CMD-SHELL
        - pg_isready -U mexibanco -d mexibanco
      timeout: 3s
      interval: 5s
      retries: 5
    image: postgres:17-alpine
    networks:
      default: null
    volumes:
      - type: volume
        source: mexibanco_datos
        target: /var/lib/postgresql/data
        volume: {}
networks:
  default:
    name: s05_santana_derek
volumes:
  mexibanco_datos:
    name: s05_santana_derek_mexibanco_datos
