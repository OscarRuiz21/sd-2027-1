Restar con condición

BEGIN;
UPDATE inventario
   SET disponibles = disponibles - 1
 WHERE producto = 'consola' AND disponibles > 0;
-- 0 filas: ROLLBACK y responder «se agotaron»
-- 1 fila: guardar el apartado
INSERT INTO apartados (id, cuenta, estado, vence_en)
VALUES (:id, :cuenta, 'apartado',
